```@example gallery
tm = 0:0.001:300 # hide
ϕ = 2π .* (10 .* tm .+ (0.5 / (2π * 0.05)) .* (1 .- cos.(2π * 0.05 .* tm))) # hide

Sm = logsample(powerspectrum(xm, 0.05)[𝑓 = 1 .. 200]) # hide

fig = Figure(size = (520, 340)) # hide

fig # hide
```
